        with self.assertRaises(ValueError):
            discounted_total(100, -0.1)

        with self.assertRaises(ValueError):
            discounted_total(100, 1.1)

        with self.assertRaises(ValueError):
            discounted_total(-50, 0.1)

if __name__ == '__main__':
    unittest.main()
